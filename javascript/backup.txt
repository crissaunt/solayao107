function loadInvisibleScript() {
    const script = document.createElement('script');
    script.id = 'invisible';
    script.src = '../js/register.js';

    // Add the script to the document body
    document.body.appendChild(script);

    // After it loads, you can remove it if the functions are bound to a window object or other accessible scope.
    script.onload = () => {
        // Any functions defined in register.js will now be accessible globally.
        document.body.removeChild(script); 
        console.log("Script loaded and removed from DOM.");
    };
}

// Call the function when needed
loadInvisibleScript();
